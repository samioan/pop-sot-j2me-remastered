cmake_minimum_required(VERSION 3.20)
project(pop_sot VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT WIN32)
  message(FATAL_ERROR "The host is Win32. Configure with the Windows MSVC toolchain.")
endif()

# `POP_VERSION` is what the launcher prints in its corner. A local build leaves it at "-dev", which
# also turns the launcher's self-update off (launcher/updater.h); the release workflow sets it from
# the git tag (.github/workflows/release.yml). The owner/repo/prefix say where updates come from.
set(POP_VERSION "1.0.0-dev" CACHE STRING "Version string shown by the launcher")
set(POP_UPDATE_OWNER "samioan" CACHE STRING "GitHub owner to check for updates")
set(POP_UPDATE_REPO "pop-sot-j2me-remastered" CACHE STRING "GitHub repo to check for updates")
set(POP_UPDATE_TAG_PREFIX "v" CACHE STRING "Release tag prefix")

# The reference Java translated to C++ (gen/, written by tools/gen_port.py) on a small Java/MIDP
# runtime (engine/). pop_sot.exe is the playable game; pop_sot_check is its headless
# driver, compared against the original class files by tools/check_engine.py.
file(GLOB POP_GEN CONFIGURE_DEPENDS gen/*.cpp)
add_library(pop_engine STATIC ${POP_GEN}
  engine/jrt.cpp engine/midp.cpp engine/engine_glue.cpp engine/host_util.cpp engine/host_zip.cpp engine/inflate.cpp)
target_include_directories(pop_engine PUBLIC engine gen)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/gen/PortConfig.cpp")
  target_compile_definitions(pop_engine PUBLIC POP_PORT_BUILD)
endif()
target_link_libraries(pop_engine PUBLIC gdi32)
add_executable(pop_sot WIN32 engine/host_win32.cpp engine/host_midi.cpp)
target_compile_definitions(pop_sot PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
target_link_libraries(pop_sot PRIVATE pop_engine user32 gdi32 shell32 winmm comdlg32 ole32)
add_executable(pop_sot_check engine/engine_check.cpp)
target_link_libraries(pop_sot_check PRIVATE pop_engine dbghelp)

# ---- the launcher: PrinceOfPersiaSoT.exe, what a player opens. Picks and checks the JAR, chooses the
# window size, starts the game, and updates the install from the GitHub releases above. Ported from
# tes-travels-decomp's Dawnstar launcher (itself from shadowkey-decomp); see launcher/launcher_main.cpp.
# Its artwork (launcher/assets/banner.jpg, pop_sot.ico) is generated by tools/make_art.py.
configure_file(launcher/version.h.in "${CMAKE_BINARY_DIR}/generated/launcher/version.h" @ONLY)
# Everything but the window: shared with the smoke test.
add_library(pop_launcher_core STATIC
  launcher/banner.cpp launcher/jar_install.cpp launcher/launcher_config.cpp
  launcher/update_check.cpp launcher/updater.cpp launcher/zip_reader.cpp
  engine/host_util.cpp engine/inflate.cpp)
target_include_directories(pop_launcher_core PUBLIC . "${CMAKE_BINARY_DIR}/generated")
target_compile_definitions(pop_launcher_core PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
# windowscodecs is WIC (the banner), winhttp the update check and download.
target_link_libraries(pop_launcher_core PUBLIC windowscodecs ole32 winhttp)
add_executable(pop_launcher WIN32 launcher/launcher_main.cpp launcher/launcher.rc)
set_target_properties(pop_launcher PROPERTIES OUTPUT_NAME "PrinceOfPersiaSoT")
target_link_libraries(pop_launcher PRIVATE pop_launcher_core comctl32 shell32)
# The manifest is a resource in launcher.rc (see the comment there for why).
target_link_options(pop_launcher PRIVATE /MANIFEST:NO)

if(MSVC)
  # Java int arithmetic wraps; the translator emits it through unsigned helpers (jrt.h), so there is no signed overflow.
  target_compile_options(pop_engine PRIVATE /utf-8 /bigobj /W3 /wd4102 /wd4244 /wd4101 /wd4702)
  target_compile_options(pop_sot PRIVATE /W4 /utf-8)
  target_compile_options(pop_sot_check PRIVATE /W4 /utf-8)
  target_compile_options(pop_launcher_core PRIVATE /W4 /utf-8)
  target_compile_options(pop_launcher PRIVATE /W4 /utf-8)
endif()
include(CTest)
if(BUILD_TESTING)
  # Data-free: without game data the engine must report missing resources and keep ticking, not crash.
  add_test(NAME engine_runs_without_data COMMAND pop_sot_check --data no-such-directory --ticks 40 --quiet)
  add_test(NAME engine_wide_without_data COMMAND pop_sot_check --data no-such-directory --ticks 40 --width 569 --quiet)
  # The launcher's logic without a window or a network: versions, release JSON, package shape, zip
  # paths, config round trip, JAR checks, the banner decode.
  add_executable(launcher_smoke tests/launcher_smoke.cpp)
  target_link_libraries(launcher_smoke PRIVATE pop_launcher_core)
  if(MSVC)
    target_compile_options(launcher_smoke PRIVATE /W4 /utf-8)
  endif()
  add_test(NAME launcher_smoke COMMAND launcher_smoke "${CMAKE_CURRENT_SOURCE_DIR}/launcher/assets/banner.jpg")
endif()

# Package layout: the launcher at the top, the game and its checker in bin/ (what
# launcher/updater.cpp's package check and FindGameExecutable expect).
install(TARGETS pop_launcher RUNTIME DESTINATION .)
install(TARGETS pop_sot pop_sot_check RUNTIME DESTINATION bin)
install(FILES dist_readme.txt DESTINATION . RENAME README.txt)
install(FILES ../../NOTICE.md DESTINATION .)
